    function killAdModals() {
      let hit = false;
      AD_MODAL_IDS.forEach((id) => {
        const el = document.getElementById(id);
        if (!el || el.dataset.jiuseKilled === '1') return;
        try {

            if (btn) {
              el.dataset.jiuseKilled = '1';
              btn.click();
              hit = true;
              log('dismiss ad modal', id);
              return;
            }
          }
          el.remove();
          hit = true;
          log('remove ad modal', id);
        } catch (_) {}
      });
      return hit;
    }

    /** document-start 守卫：在 app.js 的 ready 回调之前就把进站弹窗摘掉 */

            try {

    /**

     * Bootstrap 显示/隐藏各有 300ms 过渡，过渡中也是这个状态，所以要连续两次判定才动手。
     */
    function unstickPage() {
      try {
        if (!document.body) return;
        const shown = Array.prototype.some.call(document.querySelectorAll('.modal'), (m) => {
          if (!m.classList.contains('show')) return false;
          const cs = getComputedStyle(m);
          return cs.display !== 'none' && cs.visibility !== 'hidden';
        });
        const backdrops = document.querySelectorAll('.modal-backdrop');
        const locked = document.body.classList.contains('modal-open');
        if (shown || (!backdrops.length && !locked)) {
          unstickPage._strikes = 0;
          return;
        }
        unstickPage._strikes = (unstickPage._strikes || 0) + 1;
        if (unstickPage._strikes < 2) return;
        unstickPage._strikes = 0;

        backdrops.forEach((b) => {
          try {
            b.remove();
          } catch (_) {}
        });
        document.body.classList.remove('modal-open');
        document.body.style.removeProperty('padding-right');
        document.body.style.removeProperty('overflow');
        // 半开状态的弹窗（display:block 但没 .show）收回去
        document.querySelectorAll('.modal').forEach((m) => {
          if (m.classList.contains('show')) return;
          if (m.style && m.style.display === 'block') {
            m.style.display = 'none';
            m.style.removeProperty('padding-right');
            m.setAttribute('aria-hidden', 'true');
            m.removeAttribute('aria-modal');
          }
        });
        log('unstick page: 清掉残留遮罩 / body.modal-open');
      } catch (_) {}
    }

    function patchElementSetters() {
      const rawSetAttribute = Element.prototype.setAttribute;
      Element.prototype.setAttribute = function (name, value) {
        try {
          if (
            typeof name === 'string' &&
            typeof value === 'string' &&
            /^(src|href|data-src)$/i.test(name) &&
            isAdUrl(value)

            log('block setAttribute', name, value.slice(0, 100));
            return;

          if (typeof name === 'string' && /^(src|data-src)$/i.test(name) && isM3u8Url(String(value))) {
            captureStream(String(value));
          }
        } catch (_) {}
        return rawSetAttribute.apply(this, arguments);
      };

      for (const Tag of [HTMLScriptElement, HTMLIFrameElement, HTMLImageElement, HTMLSourceElement, HTMLVideoElement]) {
        if (!Tag || !Tag.prototype) continue;
        try {
          const desc = Object.getOwnPropertyDescriptor(Tag.prototype, 'src');
          if (!desc || !desc.set) continue;
          Object.defineProperty(Tag.prototype, 'src', {
            configurable: true,
            enumerable: desc.enumerable,
            get: desc.get,
            set(v) {
              const s = String(v);
              if (isM3u8Url(s)) captureStream(s);
              if (isAdUrl(s)) {
                log('block src', Tag.name, s.slice(0, 100));
                return;
              }
              return desc.set.call(this, s);
            },
          });
        } catch (e) {
          log('patch src failed', Tag.name, e);
        }
      }
    }

            log('fetch patch error', e);

              if (typeof self.onload === 'function') self.onload();
            } catch (_) {}
          }, 0);
          return;

        return XS.apply(this, arguments);
      };

              e.stopImmediatePropagation();

          }
        } catch (_) {}
      };
      ['click', 'auxclick', 'mousedown', 'pointerdown'].forEach((type) => {
        document.addEventListener(type, blockAdClick, true);
      });
    }

          return false;
        }
        if (el.closest && el.closest('#video-play, .videoPlayContainer, .video-js, #' + PANEL_ID)) {
          return false;
        }
        el.remove();
        return true;
      } catch (_) {
        return false;
      }
    }

    function isExternalAdLink(a) {
      if (!a || !a.getAttribute) return false;
      const href = a.href || a.getAttribute('href') || '';
      if (!href || href.startsWith('#') || href.startsWith('/') || href.startsWith('mailto:')) {
        // 相对路径可能是站内
        if (href.startsWith('/') || href.startsWith('mailto:') || href.startsWith('#')) return false;
      }
      try {
        if (isSameSite(href)) return false;
        if (isAdUrl(href)) return true;
        if (a.target === '_blank' && /nofollow/i.test(a.rel || '')) return true;
        // 纯外链且不是常见站内镜像
        const h = hostOf(href);
        if (h && !KEEP_HOST_RE.test(h) && !isSameSite(href)) {
          // 推荐区真实视频是 /video/view/ 同源；外链即广告
          return true;
        }
      } catch (_) {}
      return false;
    }

    function isAdNode(el) {
      if (!el || el.nodeType !== 1) return false;
      const tag = el.tagName;
      const id = el.id || '';
      const cls = typeof el.className === 'string' ? el.className : String(el.className || '');

      if (id === PANEL_ID || id === STYLE_ID) return false;
      if (id === 'video-play' || /videoPlayContainer|video-js|vjs-/i.test(cls)) return false;
      if (isModalChrome(el) || inSiteModal(el)) return false;

      // 文字广告条
      if (/\balert\b/i.test(cls) && (/\balert-warning\b|\balert-danger\b/i.test(cls))) {
        const a = el.querySelector && el.querySelector('a[href]');
        if (a && isExternalAdLink(a)) return true;
      }

      // 假视频卡片：video-elem 指向外站
      if (/\bvideo-elem\b/i.test(cls)) {
        const a = el.querySelector && el.querySelector('a[href]');
        if (a && isExternalAdLink(a)) return true;
      }

      if (tag === 'SCRIPT') return isAdUrl(el.src || el.getAttribute('src') || '');
      if (tag === 'IFRAME' || tag === 'IMG') return isAdUrl(el.src || el.getAttribute('src') || '');
      if (tag === 'A') return isExternalAdLink(el);

      // 哈希广告容器：内部几乎全是外链
      // 只认「首字母+15位十六进制」，否则 registerForm / noticeModal 这类正常 id 会被当成广告
      if (id && HASH_ID_RE.test(id) && el.querySelectorAll) {
        const links = el.querySelectorAll('a[href]');
        if (links.length >= 2) {
          let ext = 0;
          links.forEach((a) => {
            if (isExternalAdLink(a)) ext += 1;
          });
          if (ext >= 2 && ext === links.length) return true;
        }
      }
      return false;
    }

    }

    /**
     * 高清VIP页面：从普通页面抓取真实 m3u8 流地址，
     * 替换掉服务器下发的固定预览流 (cdn2.jiuse2.cloud/hlsd/js10/index.m3u8)
     * 实现免登录看高清正片
     */
    async function replaceHdPreviewWithReal() {
      if (!isHdPreviewPage()) return;
      // 已完成替换，不再重复执行（每次 fetch 的 t= token 不同，不能用 URL 相等判断）
      if (replaceHdPreviewWithReal._done === location.pathname) return;

      try {
        const videoId = readVideoId();
        if (!videoId) return;

        const resp = await fetch('/video/view/' + videoId);
        const html = await resp.text();
        const srcMatch = html.match(/data-src="(https?:)?(\/\/[^"]*m3u8[^"]*)"/);
        const realM3u8 = srcMatch
          ? (srcMatch[1] || 'https:') + srcMatch[2].replace(/&amp;/g, '&')
          : '';
        if (!realM3u8 || realM3u8.includes('js10/index.m3u8')) return;

        const videoPlay = document.getElementById('video-play');
        if (!videoPlay) return;

        // 去掉 token 参数比较路径本体，防止 fetch 获得新 token 后重复触发
        const curSrc = videoPlay.getAttribute('data-src') || '';
        const curPath = curSrc.replace(/[?&]t=\d+&m=[^&]+/, '').replace(/[?&]t=\d+/, '');
        const newPath = realM3u8.replace(/[?&]t=\d+&m=[^&]+/, '').replace(/[?&]t=\d+/, '');
        if (curPath === newPath) return;

        if (videoPlay.querySelector('video') && !(videoPlay.querySelector('video').src || '').includes('js10')) {
          // 视频已是非预览流，不再重复设置
          return;
        }

        log('HD: 替换预览流为真实正片', newPath);
        replaceHdPreviewWithReal._done = location.pathname;

        videoPlay.setAttribute('data-src', realM3u8);
        const video = videoPlay.querySelector('video');
        if (video) {
          video.setAttribute('data-src', realM3u8);
          video.src = realM3u8;
        }

        // 通知 Video.js 切换源
        if (typeof videojs !== 'undefined' && videojs.getPlayer) {
          try {
            const player = videojs.getPlayer('video-play');
            if (player) {
              player.src({ type: 'application/x-mpegURL', src: realM3u8 });
            }
          } catch (_) {}
        }

        // 同步到状态以启用下载
        state.m3u8 = realM3u8;
        updatePanelInfo();
      } catch (e) {
        log('replaceHdPreviewWithReal error', e);
      }
    }

    function sweep() {
      injectCSS();
      killAdModals();
      replaceHdPreviewWithReal();
      probeDomStreams();
      readMetaIntoState();
      try {
        // 选择器批量
        REMOVE_SELECTORS.forEach((sel) => {
          try {
            document.querySelectorAll(sel).forEach((el) => removeNode(el));
          } catch (_) {}
        });

        // 外链广告 alert 整行
        document.querySelectorAll('.alert').forEach((el) => {
          try {
            const a = el.querySelector('a[href]');
            if (!a || !isExternalAdLink(a)) return;
            // 弹窗里的 alert 常常同时是 .modal-body，只摘链接
            if (isModalChrome(el) || inSiteModal(el)) {
              stripAdLinks(el);
              return;
            }
            {
              const row = el.closest('.row') || el.closest('.col-sm') || el;
              // 若 row 内全是广告则删 row
              if (row && row !== el) {
                const links = row.querySelectorAll('a[href]');
                let allAd = links.length > 0;
                links.forEach((l) => {
                  if (!isExternalAdLink(l)) allAd = false;
                });
                removeNode(allAd ? row : el);
              } else {
                removeNode(el);
              }
            }
          } catch (_) {}
        });

        // 外链假视频卡片
        document.querySelectorAll('.video-elem').forEach((el) => {
          try {
            const a = el.querySelector('a[href]');
            if (a && isExternalAdLink(a)) {
              const col = el.closest('.col-30, .col-sm-20, .col-md-15, [class*="col-"]') || el;
              removeNode(col);
            }
          } catch (_) {}
        });

        // 站外图片广告块（col-sm 内只有一张外链图）
        document.querySelectorAll('a[target="_blank"][rel*="nofollow"], a[target="_blank"][href^="http"]').forEach((a) => {
          try {
            if (!isExternalAdLink(a)) return;
            // 不在播放器/面板内
            if (a.closest && a.closest('#video-play, .videoPlayContainer, #' + PANEL_ID)) return;
            // 站内弹窗（警告 / 登录 / 支付…）：只摘链接本身，结构一律不动
            if (inSiteModal(a)) {
              removeNode(a);
              return;
            }
            // 详情「关于/简介」区：只删外链本身，避免整段简介被干掉
            if (a.closest && a.closest('#videoShowTabAbout, #videoShowTabDownload, #videoShowTabShare, .videoInfos, .container-title')) {
              removeNode(a);
              return;
            }
            const col = a.closest('.col-sm, .col-30, [class*="col-"]');
            const host = col || a;
            // 若父级是广告列表容器则尽量删父级单元
            removeNode(host);
          } catch (_) {}
        });

        // 脚本/iframe 广告
        document.querySelectorAll('script[src], iframe[src]').forEach((el) => {
          if (isAdUrl(el.src)) removeNode(el);
        });

        // 关闭移动端广告层
        document.querySelectorAll('button[aria-label="Close"]').forEach((btn) => {
          try {
            const t = (btn.textContent || '').trim();
            if (t === '关闭' || /mabs/i.test(btn.getAttribute('onclick') || '')) {
              btn.click();
            }
          } catch (_) {}
        });
      } catch (e) {
        log('sweep error', e);
      }
      unstickPage();
      updatePanelInfo();
    }

    function observe() {
      const mo = new MutationObserver((mutations) => {
        let need = false;
        for (const m of mutations) {
          if (m.type === 'childList') {
            m.addedNodes.forEach((n) => {
              if (n.nodeType !== 1) return;
              if (isAdNode(n)) {
                removeNode(n);
                return;
              }
              if (n.querySelectorAll) {
                // 新插入的 data-src
                if (n.matches && n.matches('[data-src*="m3u8"], video, #video-play')) probeDomStreams();
                n.querySelectorAll('a[target="_blank"], .alert, .video-elem, script[src], iframe[src], [data-src*="m3u8"]').forEach(
                  (el) => {
                    if (el.dataset && el.dataset.src && isM3u8Url(el.dataset.src)) {
                      captureStream(el.dataset.src);
                    }
                    if (isAdNode(el) || (el.src && isAdUrl(el.src))) {
                      removeNode(el.closest('.col-sm, .video-elem, .row') || el);
                    }
                  }
                );
              }
              need = true;
            });
          } else if (m.type === 'attributes') {
            const el = m.target;
            if (!el) continue;
            if (m.attributeName === 'src' || m.attributeName === 'data-src' || m.attributeName === 'href') {
              const val = el.getAttribute(m.attributeName) || el.src || el.href || '';
              if (isM3u8Url(val)) captureStream(val);
              if (isAdUrl(val)) removeNode(el);
            }
          }
        }
        if (need) {
          // 节流：合并到下一帧
          if (!observe._raf) {
            observe._raf = requestAnimationFrame(() => {
              observe._raf = 0;
              probeDomStreams();
              readMetaIntoState();
              updatePanelInfo();
            });
          }
        }
      });
      mo.observe(document.documentElement || document, {
        childList: true,
        subtree: true,
        attributes: true,
        attributeFilter: ['src', 'href', 'data-src', 'data-poster', 'class'],
      });
    }

    function sanitizeFilename(name) {
      return (
        String(name || 'jiuse')

    function formatBytes(n) {
      if (!n || n < 0) return '0 B';
      const u = ['B', 'KB', 'MB', 'GB'];
      let i = 0;
      let v = n;
      while (v >= 1024 && i < u.length - 1) {
        v /= 1024;
        i += 1;
      }
      return v.toFixed(i ? 1 : 0) + ' ' + u[i];
    }

    function hexToBytes(hex) {
      const h = String(hex || '').replace(/^0x/i, '').replace(/\s+/g, '');
      const out = new Uint8Array(h.length / 2);
      for (let i = 0; i < out.length; i++) out[i] = parseInt(h.substr(i * 2, 2), 16);
      return out;
    }

    function seqToIv(sn) {
      const iv = new Uint8Array(16);
      let n = sn >>> 0;
      iv[15] = n & 0xff;
      iv[14] = (n >>> 8) & 0xff;
      iv[13] = (n >>> 16) & 0xff;
      iv[12] = (n >>> 24) & 0xff;
      return iv;
    }

    function parseM3u8(text, baseUrl) {
      const lines = String(text || '').split(/\r?\n/);
      const segs = [];
      let keyUri = '';
      let method = 'NONE';
      let ivBytes = null;
      let mediaSequence = 0;
      let expectUri = false;

      const sawMediaSeq = lines.some((l) => /^#EXT-X-MEDIA-SEQUENCE:/i.test(l.trim()));
      if (sawMediaSeq) {
        const m = String(text).match(/#EXT-X-MEDIA-SEQUENCE:(\d+)/i);
        if (m) mediaSequence = parseInt(m[1], 10) || 0;
      }

      let sn = mediaSequence;
      for (let i = 0; i < lines.length; i++) {
        const line = lines[i].trim();
        if (!line) continue;
        if (line.startsWith('#')) {
          if (/^#EXT-X-KEY:/i.test(line)) {
            const mu = line.match(/METHOD=([^,]+)/i);
            method = mu ? mu[1].trim() : 'NONE';
            const uu = line.match(/URI="([^"]+)"/i) || line.match(/URI=([^,]+)/i);
            keyUri = uu ? uu[1].trim() : '';
            const ivm = line.match(/IV=0x([0-9a-fA-F]+)/i);
            ivBytes = ivm ? hexToBytes(ivm[1]) : null;
          } else if (/^#EXTINF:/i.test(line)) {
            expectUri = true;
          }
          continue;
        }
        if (expectUri || !line.startsWith('#')) {
          let href = line;
          try {
            href = new URL(line, baseUrl).href;
          } catch (_) {}
          segs.push({
            url: href,
            sn,
            method,
            keyUri,
            iv: ivBytes ? new Uint8Array(ivBytes) : seqToIv(sn),
          });
          sn += 1;
          expectUri = false;
        }
      }
      return { segs, mediaSequence };
    }

    async function fetchBuf(url, signal) {
      const res = await fetch(url, { credentials: 'omit', signal });
      if (!res.ok) throw new Error('HTTP ' + res.status + ' ' + url.slice(0, 80));
      return res.arrayBuffer();
    }

    async function decryptAes128(data, keyBytes, ivBytes) {
      const cryptoKey = await crypto.subtle.importKey('raw', keyBytes, { name: 'AES-CBC' }, false, ['decrypt']);
      return await crypto.subtle.decrypt({ name: 'AES-CBC', iv: ivBytes }, cryptoKey, data);
    }

    async function downloadHls(m3u8Url, { onProgress, signal } = {}) {
      const listRes = await fetch(m3u8Url, { credentials: 'omit', signal });
      if (!listRes.ok) throw new Error('获取 m3u8 失败 HTTP ' + listRes.status);
      const text = await listRes.text();
      if (!/#EXTM3U/i.test(text)) throw new Error('不是有效的 m3u8');

      if (/#EXT-X-STREAM-INF/i.test(text)) {
        const lines = text.split(/\r?\n/).map((l) => l.trim());
        let best = null;
        let lastBw = 0;
        for (let i = 0; i < lines.length; i++) {
          const line = lines[i];
          if (/^#EXT-X-STREAM-INF/i.test(line)) {
            const bm = line.match(/BANDWIDTH=(\d+)/i);
            const bw = bm ? parseInt(bm[1], 10) : 0;
            const next = lines[i + 1];
            if (next && !next.startsWith('#') && bw >= lastBw) {
              lastBw = bw;
              try {
                best = new URL(next, m3u8Url).href;
              } catch (_) {
                best = next;
              }
            }
          }
        }
        if (!best) throw new Error('主播放列表中未找到子流');
        return downloadHls(best, { onProgress, signal });
      }

      const { segs } = parseM3u8(text, m3u8Url);
      if (!segs.length) throw new Error('播放列表无分片');

      const keyCache = new Map();
      async function getKey(uri) {
        if (!uri) return null;
        let abs = uri;
        try {
          abs = new URL(uri, m3u8Url).href;
        } catch (_) {}
        if (keyCache.has(abs)) return keyCache.get(abs);
        const buf = await fetchBuf(abs, signal);
        const bytes = new Uint8Array(buf);
        keyCache.set(abs, bytes);
        return bytes;
      }

      const total = segs.length;
      const parts = new Array(total);
      let done = 0;
      let bytes = 0;
      let idx = 0;
      let failed = null;

      async function worker() {
        while (idx < total) {
          if (signal && signal.aborted) throw new DOMException('已取消', 'AbortError');
          if (failed) throw failed;
          const i = idx++;
          const seg = segs[i];
          let attempt = 0;
          while (attempt < 3) {
            try {
              let buf = await fetchBuf(seg.url, signal);
              if (seg.method && /AES-128/i.test(seg.method) && seg.keyUri) {
                const key = await getKey(seg.keyUri);
                const iv = seg.iv || seqToIv(seg.sn);
                buf = await decryptAes128(buf, key, iv);
              }
              parts[i] = buf;
              bytes += buf.byteLength;
              done += 1;
              if (onProgress) onProgress({ done, total, bytes, index: i });
              break;
            } catch (e) {
              if (e && e.name === 'AbortError') throw e;
              attempt += 1;
              if (attempt >= 3) {
                failed = new Error('分片失败 #' + i + ': ' + (e && e.message));
                throw failed;
              }
              await new Promise((r) => setTimeout(r, 400 * attempt));
            }
          }
        }
      }

      const workers = [];
      const n = Math.min(DL_CONCURRENCY, total || 1);
      for (let i = 0; i < n; i++) workers.push(worker());
      await Promise.all(workers);

      return new Blob(parts, { type: 'video/mp2t' });
    }

    async function triggerBlobDownload(blob, filename) {
      const url = URL.createObjectURL(blob);
      try {
        const a = document.createElement('a');
        a.href = url;
        a.download = filename;
        a.rel = 'noopener';
        a.style.cssText = 'position:fixed;left:-9999px;top:-9999px';
        document.documentElement.appendChild(a);
        a.click();
        setTimeout(() => {
          try {
            a.remove();
          } catch (_) {}
        }, 2000);
      } catch (e) {
        log('a.download failed', e);
      }

      if (IS_MOBILE) {
        try {
          const file = new File([blob], filename, { type: blob.type || 'video/mp2t' });
          if (navigator.canShare && navigator.canShare({ files: [file] }) && navigator.share) {
            await navigator.share({ files: [file], title: filename, text: filename });
            setTimeout(() => {
              try {
                URL.revokeObjectURL(url);
              } catch (_) {}
            }, 60000);
            return 'share';
          }
        } catch (e) {
          if (e && e.name === 'AbortError') {
            setTimeout(() => {
              try {
                URL.revokeObjectURL(url);
              } catch (_) {}
            }, 60000);
            return 'share-cancel';
          }
          log('share failed', e);
        }
        try {
          const opened = window.open(url, '_blank');
          if (!opened) setStatus('请允许弹窗，或改用「复制流」到下载器', 'wait');
        } catch (_) {}
      }

      setTimeout(() => {
        try {
          URL.revokeObjectURL(url);
        } catch (_) {}
      }, IS_MOBILE ? 120000 : 15000);
      return 'download';
    }

    async function copyText(text) {
      try {
        if (navigator.clipboard && navigator.clipboard.writeText) {
          await navigator.clipboard.writeText(text);
          return true;
        }
      } catch (_) {}
      try {
        const ta = document.createElement('textarea');
        ta.value = text;
        ta.setAttribute('readonly', '');
        ta.style.cssText =
          'position:fixed;top:0;left:0;width:1px;height:1px;padding:0;margin:0;border:0;opacity:0;';
        document.body.appendChild(ta);
        ta.focus();
        ta.setSelectionRange(0, ta.value.length);
        const ok = document.execCommand('copy');
        ta.remove();
        return ok;
      } catch (_) {
        return false;
      }
    }

    function qs(root, sel) {
      return root.querySelector(sel);
    }

    function setPanelState(mode) {
      const panel = document.getElementById(PANEL_ID);
      if (!panel) return;
      const map = {
        wait: '探测中',
        ready: '可下载',
        busy: '下载中',
        ok: '已完成',
        err: '出错',
        idle: '待命',
      };
      const key = mode || (state.m3u8 ? 'ready' : isDetailPage() ? 'wait' : 'idle');
      panel.dataset.state = key === 'idle' ? 'wait' : key;
      panel.classList.toggle('is-busy', key === 'busy');
      const badge = qs(panel, '[data-el="badge"]');
      if (badge) badge.textContent = map[key] || map.wait;
    }

    function setStatus(text, mode) {
      const panel = document.getElementById(PANEL_ID);
      if (!panel) return;
      const el = qs(panel, '[data-el="status"]');
      if (el) el.textContent = text || '';
      if (mode) setPanelState(mode);
    }

    function setProgress(pct) {
      const panel = document.getElementById(PANEL_ID);
      if (!panel) return;
      const val = Math.max(0, Math.min(100, pct || 0));
      const bar = qs(panel, '[data-el="bar"]');
      const label = qs(panel, '[data-el="pct"]');
      if (bar) bar.style.width = val + '%';
      if (label) label.textContent = Math.round(val) + '%';
    }

    function shortUrl(url) {
      if (!url) return '';
      try {
        const u = new URL(url);
        const path = u.pathname.split('/').filter(Boolean).slice(-2).join('/') || u.pathname;
        return u.hostname.replace(/^www\./, '') + '/' + path + (u.search ? '…' : '');
      } catch (_) {
        return String(url).slice(0, 64);
      }
    }

    function updatePanelInfo() {
      const panel = document.getElementById(PANEL_ID);
      if (!panel) return;
      readMetaIntoState();
      const t = qs(panel, '[data-el="title"]');
      const u = qs(panel, '[data-el="url"]');
      const vid = qs(panel, '[data-el="vid"]');
      const authorEl = qs(panel, '[data-el="author"]');
      const fnameEl = qs(panel, '[data-el="fname"]');
      const titleText = state.title || (isDetailPage() ? '视频详情页' : '请先打开视频详情页');
      if (t) {
        t.textContent = titleText;
        t.title = titleText;
      }
      if (authorEl) authorEl.textContent = state.author || (isDetailPage() ? '读取中…' : '—');
      if (fnameEl) fnameEl.textContent = buildFilename();
      if (u) {
        if (state.m3u8) {
          u.textContent = shortUrl(state.m3u8);
          u.title = state.m3u8;
        } else {
          u.textContent = isDetailPage() ? '等待播放器加载流地址…' : '进入详情页后自动捕获';
          u.title = '';
        }
      }
      if (vid) vid.textContent = state.videoId ? '#' + state.videoId : '';
      const dl = qs(panel, '[data-act="download"]');
      if (dl) dl.disabled = !state.m3u8 || state.downloading;
      panel.querySelectorAll('[data-act="copy"], [data-act="open"]').forEach((btn) => {
        btn.disabled = !state.m3u8 || state.downloading;
      });
      if (!state.downloading) {
        setPanelState(state.m3u8 ? 'ready' : isDetailPage() ? 'wait' : 'idle');
      }
    }

    function bindPanel(panel) {
      panel.addEventListener('click', (e) => {
        if (!panel.classList.contains('jiuse-collapsed')) return;
        if (panel.dataset.jiuseDragged === '1') return;
        if (e.target && e.target.closest && e.target.closest('[data-act="collapse"]')) return;
        if (e.target && e.target.closest && e.target.closest('[data-act="refresh"]')) return;
        const btn = panel.querySelector('[data-act="collapse"]');
        if (btn) btn.click();
      });

      panel.addEventListener('click', async (e) => {
        const btn = e.target && e.target.closest && e.target.closest('[data-act]');
        if (!btn || btn.disabled) return;
        if (panel.dataset.jiuseDragged === '1') return;
        e.preventDefault();
        e.stopPropagation();
        const act = btn.getAttribute('data-act');
        if (act === 'collapse') {
          const collapsed = panel.classList.toggle('jiuse-collapsed');
          if (IS_MOBILE) {
            panel.classList.toggle('jiuse-sheet', !collapsed);
            if (!collapsed) {
              panel.style.left = '';
              panel.style.top = '';
              panel.style.right = '';
              panel.style.bottom = '';
            }
          }
          const iconBtn = panel.querySelector('[data-act="collapse"]');
          if (iconBtn) iconBtn.innerHTML = collapsed ? ICONS.plus : ICONS.minus;
          return;
        }
        if (act === 'refresh') {
          btn.classList.add('is-spinning');
          setTimeout(() => btn.classList.remove('is-spinning'), 700);
          state.m3u8 = '';
          probeDomStreams();
          readMetaIntoState();
          updatePanelInfo();
          setStatus(state.m3u8 ? '已重新捕获流地址' : '未找到流，请先点击播放', state.m3u8 ? 'ready' : 'wait');
          return;
        }
        if (act === 'copy') {
          if (!state.m3u8) return;
          const ok = await copyText(state.m3u8);
          setStatus(ok ? '已复制 m3u8 地址' : '复制失败，请手动选择', ok ? 'ok' : 'err');
          return;
        }
        if (act === 'copy-page') {
          const ok = await copyText(location.href);
          setStatus(ok ? '已复制页面链接' : '复制失败', ok ? 'ok' : 'err');
          return;
        }
        if (act === 'open') {
          if (!state.m3u8) return;
          try {
            window.open(state.m3u8, '_blank', 'noopener');
            setStatus('已在新标签打开 m3u8', 'ok');
          } catch (_) {
            setStatus('打开失败', 'err');
          }
          return;
        }
        if (act === 'stop') {
          if (state.abort) {
            try {
              state.abort.abort();
            } catch (_) {}
          }
          return;
        }
        if (act === 'download') {
          await startDownload();
        }
      });
    }

    async function startDownload() {
      if (state.downloading) return;
      if (!state.m3u8) {
        probeDomStreams();
        if (!state.m3u8) {
          setStatus('尚未捕获到视频流，请先播放视频', 'err');
          return;
        }
      }
      readMetaIntoState();
      const filename = buildFilename();
      state.downloading = true;
      state.abort = typeof AbortController !== 'undefined' ? new AbortController() : null;
      setProgress(0);
      setStatus('开始下载…', 'busy');
      updatePanelInfo();

      try {
        const blob = await downloadHls(state.m3u8, {
          signal: state.abort ? state.abort.signal : undefined,
          onProgress: ({ done, total, bytes }) => {
            const pct = total ? (done / total) * 100 : 0;
            setProgress(pct);
            setStatus(`下载分片 ${done}/${total} · ${formatBytes(bytes)}`, 'busy');
          },
        });
        setProgress(100);
        setStatus(`合并完成 ${formatBytes(blob.size)}，正在保存…`, 'busy');
        const mode = await triggerBlobDownload(blob, filename);
        setStatus(
          mode === 'share' || mode === 'share-cancel'
            ? `已处理：${filename}`
            : `已下载：${filename}`,
          'ok'
        );
      } catch (e) {
        if (e && e.name === 'AbortError') {
          setStatus('已取消下载', 'wait');
          setProgress(0);
        } else {
          log('download failed', e);
          setStatus('下载失败：' + (e && e.message ? e.message : String(e)), 'err');
        }
      } finally {
        state.downloading = false;
        state.abort = null;
        updatePanelInfo();
      }
    }

    function makeDraggable(box, handle) {
      if (!box || !handle) return;
      let ox = 0;
      let oy = 0;
      let sx = 0;
      let sy = 0;
      let dragging = false;
      let moved = false;

      const onDown = (e) => {
        if (e.type === 'mousedown' && e.button !== 0) return;
        // 按钮不拖
        if (e.target && e.target.closest && e.target.closest('button, a, input')) return;
        const pt = e.touches ? e.touches[0] : e;
        const rect = box.getBoundingClientRect();
        dragging = true;
        moved = false;
        box.dataset.jiuseDragged = '0';
        box.classList.add('is-dragging');
        sx = pt.clientX;
        sy = pt.clientY;
        ox = rect.left;
        oy = rect.top;
        box.style.right = 'auto';
        box.style.bottom = 'auto';
        box.style.left = ox + 'px';
        box.style.top = oy + 'px';
        e.preventDefault();
      };

      const onMove = (e) => {
        if (!dragging) return;
        const pt = e.touches ? e.touches[0] : e;
        const dx = pt.clientX - sx;
        const dy = pt.clientY - sy;
        if (Math.abs(dx) + Math.abs(dy) > 4) moved = true;
        const w = box.offsetWidth;
        const h = box.offsetHeight;
        let nx = ox + dx;
        let ny = oy + dy;
        const maxX = window.innerWidth - Math.min(w, window.innerWidth);
        const maxY = window.innerHeight - Math.min(h, 80);
        nx = Math.max(0, Math.min(maxX, nx));
        ny = Math.max(0, Math.min(Math.max(0, maxY), ny));
        box.style.left = nx + 'px';
        box.style.top = ny + 'px';
        if (e.cancelable) e.preventDefault();
      };

      const onUp = () => {
        if (!dragging) return;
        dragging = false;
        box.classList.remove('is-dragging');
        if (moved) {
          box.dataset.jiuseDragged = '1';
          setTimeout(() => {
            box.dataset.jiuseDragged = '0';
          }, 80);
        }
      };

      handle.addEventListener('mousedown', onDown);
      handle.addEventListener('touchstart', onDown, { passive: false });
      window.addEventListener('mousemove', onMove, { passive: false });
      window.addEventListener('touchmove', onMove, { passive: false });
      window.addEventListener('mouseup', onUp);
      window.addEventListener('touchend', onUp);
    }

    // SPA / 站内跳转时重置
    function onRouteMaybe() {
      try {
        const key = location.pathname + location.search;
        if (onRouteMaybe._key === key) return;
        onRouteMaybe._key = key;
        state.m3u8 = '';
        state.title = '';
        state.author = '';
        state.videoId = '';
        state.poster = '';
        replaceHdPreviewWithReal._done = '';
        setTimeout(() => {
          probeDomStreams();
          readMetaIntoState();
          sweep();
          ensurePanel();
          updatePanelInfo();
        }, 200);
      } catch (_) {}
    }

    function hookHistory() {
      try {
        const wrap = (fn) =>
          function () {
            const r = fn.apply(this, arguments);
            onRouteMaybe();
            return r;
          };
        history.pushState = wrap(history.pushState.bind(history));
        history.replaceState = wrap(history.replaceState.bind(history));
        window.addEventListener('popstate', onRouteMaybe);
      } catch (_) {}
    }

    // ---------- boot ----------
    neutralizeAdGlobals();
    watchAdModalsEarly();
    patchElementSetters();
    patchNetwork();

    const bootDom = () => {
      if (bootDom._done) return;
      bootDom._done = true;
      try {
        patchWindowOpen();
        injectCSS();
        observe();
        sweep();
        ensurePanel();
        probeDomStreams();
        readMetaIntoState();
        updatePanelInfo();
        // 周期性兜底（广告动态插入 / 播放器晚加载）
        let ticks = 0;
        const timer = setInterval(() => {
          ticks += 1;
          sweep();
          replaceHdPreviewWithReal();
          probeDomStreams();
          if (ticks > 40) clearInterval(timer); // ~20s
        }, 500);
        // 轻量长驻：每 3s 探测一次流 + 清广告
        setInterval(() => {
          try {
            killAdModals();
            unstickPage();
            probeDomStreams();
            replaceHdPreviewWithReal();
            readMetaIntoState();
            // 轻扫：外链 alert
            document.querySelectorAll('.alert a[target="_blank"]').forEach((a) => {
              if (isExternalAdLink(a)) removeNode(a.closest('.alert') || a);
            });
            updatePanelInfo();
          } catch (_) {}
        }, 3000);
      } catch (e) {
        log('bootDom error', e);
      }
    };

    if (document.documentElement) injectCSS();
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', bootDom, { once: true });
    } else {
      bootDom();
    }
    // document-start 时 body 可能尚未就绪
    if (!document.body) {
      try {
        const mo = new MutationObserver(() => {
          if (document.body) {
            mo.disconnect();
            bootDom();
          }
        });
        // 观察 document 而不是 documentElement：document-start 时后者可能还是 null
        mo.observe(document, { childList: true, subtree: true });
      } catch (e) {
        log('body watch failed', e);
      }
    }

    hookHistory();
    log('loaded', location.href);
  }

  // 注入页面上下文（兼容某些扩展隔离世界）
  try {
    if (typeof unsafeWindow !== 'undefined' && unsafeWindow !== window) {
      // userscript 沙箱：尽量跑在页面
    }
  } catch (_) {}

  main();
})();
