        root.appendChild(s3);

      if (input instanceof URL) return String(input);
      if (typeof input.url === 'string') return input.url;
      if (typeof Request !== 'undefined' && input instanceof Request) return input.url;

  function isSameSite(value) {

    if (!host) return true;

  function isMediaAsset(value) {
    const u = safeUrl(value);
    if (!u) return false;
    if (!/(^|\.)(?:jerkmotion\.com|plyr\.io)$/i.test(u.hostname)) return false;
    if (/(^|\.)stats\.jerkmotion\.com$/i.test(u.hostname)) return false;
    if (/\/partners\//i.test(u.pathname)) return false;
    if (MEDIA_PATH_RE.test(u.pathname)) return true;
    if (/\.(?:mp4|webm|m3u8|vtt|jpg|jpeg|png|webp|svg|css|js|woff2?)(?:$|\?)/i.test(u.pathname)) {
      return true;
    }
    // 业务 API（视频元数据、列表、搜索）
    if (/(^|\.)api\.jerkmotion\.com$/i.test(u.hostname)) {
      if (SITE_AD_PATH_RE.test(u.pathname + u.search)) return false;
      return true;
    }
    // 页面自身 / RSC
    if (/(^|\.)jerkmotion\.com$/i.test(u.hostname) && !/(^|\.)stats\./i.test(u.hostname)) {
      if (SITE_AD_PATH_RE.test(u.pathname + u.search)) return false;
      return true;
    }
    if (/(^|\.)plyr\.io$/i.test(u.hostname)) return true;
    return false;
  }

  function isSiteAdApi(value) {
    const u = safeUrl(value);
    if (!u) return false;
    if (!/(^|\.)(?:api\.)?jerkmotion\.com$/i.test(u.hostname)) return false;
    const pathq = u.pathname + u.search;
    if (/\/partners\b/i.test(pathq)) return true;
    if (/\/m\/(?:decide|z|i)\b/i.test(pathq)) return true;
    if (/[?&]placement=/i.test(pathq) && /partners|decide/i.test(pathq)) return true;
    return false;
  }

  function isBlockedUrl(value) {
    const raw = String(value || '');
    if (!raw || raw.startsWith('data:') || raw.startsWith('blob:')) return false;
    if (raw.startsWith('about:')) return false;

    const u = safeUrl(raw);
    if (!u) return TRACKER_RE.test(raw) || BLOCK_HOST_RE.test(raw);

    // 明确放行媒体与本站业务
    if (isMediaAsset(u.href) && !isSiteAdApi(u.href)) return false;

    if (BLOCK_HOST_RE.test(u.hostname) || BLOCK_HOST_RE.test(u.href)) return true;
    if (TRACKER_RE.test(u.href)) return true;
    if (isSiteAdApi(u.href)) return true;
    if (/\/partners\//i.test(u.pathname)) return true;
    if (/(?:^|\.)stats\.jerkmotion\.com$/i.test(u.hostname)) return true;

    // 非白名单第三方
    if (!isAllowedHost(u.href) && !isSameSite(u.href)) {
      // 允许常见无害相对/协议相对已在 isSameSite 处理
      return true;
    }
    return false;
  }

  function isTracker(value) {

    if (href.startsWith('mailto:') || href.startsWith('tel:')) return false;
    return !isSameSite(href);
  }

  function hardenPrivacy() {
    try {

        enableServices() {},

        sizeMapping() {
          return { addSize() { return this; }, build() { return []; } };
        },
      });
    } catch {
      /* ignore */
    }

    // Umami / 会话录制
    try {
      const noopUmami = function umami() { /* no-op */ };
      noopUmami.track = function () {};
      noopUmami.identify = function () {};
      replaceValue(window, 'umami', noopUmami);
    } catch {
      /* ignore */
    }

    if (SETTINGS.privacyLock) {
      try {
        const rawBeacon = navigator.sendBeacon && navigator.sendBeacon.bind(navigator);
        if (rawBeacon) {
          navigator.sendBeacon = function (url) {
            try {
              if (isTracker(String(url || ''))) return true;
            } catch {
              /* ignore */
            }
            // 默认丢弃第三方 beacon
            try {
              if (!isSameSite(String(url || '')) || /stats\.jerkmotion/i.test(String(url || ''))) {
                return true;
              }
            } catch {
              return true;
            }
            return rawBeacon.apply(navigator, arguments);
          };
        }
      } catch {
        /* ignore */
      }

      try {
        blockMethod(Navigator.prototype, 'getBattery', 'battery');
        blockMethod(window.Geolocation?.prototype, 'getCurrentPosition', 'geolocation');
        blockMethod(window.Geolocation?.prototype, 'watchPosition', 'geolocation');
        blockMethod(window.MediaDevices?.prototype, 'getUserMedia', 'camera/microphone');
        blockMethod(window.MediaDevices?.prototype, 'enumerateDevices', 'media devices');
      } catch {
        /* ignore */
      }

      try {
        const blockedRtc = function blockedRtc() {
          throw securityError('WebRTC');
        };
        replaceValue(window, 'RTCPeerConnection', blockedRtc);
        if ('webkitRTCPeerConnection' in window) {
          replaceValue(window, 'webkitRTCPeerConnection', blockedRtc);
        }
      } catch {
        /* ignore */
      }

      // 清理常见追踪 cookie（保留站点必要功能）
      try {
        document.cookie.split(';').forEach((c) => {
          const name = (c.split('=')[0] || '').trim();
          if (!name) return;
          if (/^_ga|^_gid|^_gat|^__utm|^_gcl|^IDE|^TP|^umami/i.test(name)) {
            document.cookie = name + '=; expires=Thu, 01 Jan 1970 00:00:00 GMT; path=/';
            document.cookie =
              name +
              '=; expires=Thu, 01 Jan 1970 00:00:00 GMT; path=/; domain=.' +
              location.hostname.replace(/^www\./, '');
          }
        });
      } catch {
        /* ignore */
      }
    }
  }

  function patchElementSetters() {
    const rawSetAttribute = Element.prototype.setAttribute;
    Element.prototype.setAttribute = function patchedSetAttribute(name, value) {
      try {
        if (
          typeof name === 'string' &&
          typeof value === 'string' &&
          /^(src|href|data-src|data-href)$/i.test(name) &&
          isBlockedUrl(value)
        ) {
          log('block setAttribute', name, String(value).slice(0, 120));
          return;
        }
      } catch {
        /* ignore */
      }
      return rawSetAttribute.apply(this, arguments);
    };

    const patchSrc = (Tag) => {
      if (!Tag || !Tag.prototype) return;
      try {
        const desc = Object.getOwnPropertyDescriptor(Tag.prototype, 'src');
        if (!desc || !desc.set) return;
        Object.defineProperty(Tag.prototype, 'src', {
          configurable: true,
          enumerable: desc.enumerable,
          get: desc.get,
          set(v) {
            const s = String(v || '');
            if (isBlockedUrl(s)) {
              log('block src', Tag.name, s.slice(0, 120));
              return;
            }
            return desc.set.call(this, s);
          },
        });
      } catch {
        /* ignore */
      }
    };

    patchSrc(HTMLScriptElement);
    patchSrc(HTMLIFrameElement);
    patchSrc(HTMLImageElement);
    patchSrc(HTMLSourceElement);

    try {
      const hrefDesc = Object.getOwnPropertyDescriptor(HTMLAnchorElement.prototype, 'href');
      if (hrefDesc?.set) {
        Object.defineProperty(HTMLAnchorElement.prototype, 'href', {
          configurable: true,
          enumerable: hrefDesc.enumerable,
          get: hrefDesc.get,
          set(v) {

            if (isExternalNav(s) || isBlockedUrl(s)) {

            return hrefDesc.set.call(this, s);
          },
        });
      }
    } catch {
      /* ignore */
    }
  }

  function installNavigationGuard() {
    const nativeOpen = window.open?.bind(window);
    window.open = function patchedOpen(url, ...rest) {
      const raw = url == null ? '' : String(url);
      if (raw && (isExternalNav(raw) || isBlockedUrl(raw))) {
        log('block window.open', raw.slice(0, 120));
        return null;
      }
      if (!nativeOpen) return null;
      return nativeOpen(url, ...rest);
    };

    const blockExternal = (event) => {
      try {
        const anchor = event.target?.closest?.('a[href]');
        if (!anchor) return;
        if (anchor.getAttribute('data-jm-dl') === '1') return;
        if (event.target?.closest?.('#' + UI_ID)) return;

        if (!href || href.startsWith('#') || href.toLowerCase().startsWith('javascript:')) return;
        if (isExternalNav(href) || isBlockedUrl(href)) {
          event.preventDefault();
          event.stopPropagation();
          event.stopImmediatePropagation();
          log('block click', href.slice(0, 120));
        }
      } catch {
        /* ignore */
      }
    };

    ['click', 'auxclick', 'mousedown', 'pointerdown', 'touchend'].forEach((type) => {
      document.addEventListener(type, blockExternal, true);
    });
  }

  function isPartnerTile(el) {
    if (!el || el.nodeType !== 1) return false;
    try {
      // 含外链且无站内视频链接的推广块
      const links = [...el.querySelectorAll?.('a[href]') || []];
      if (!links.length && el.matches?.('a[href]')) links.push(el);
      if (!links.length) return false;
      const hasExternal = links.some((a) => isExternalNav(a.getAttribute('href') || a.href));
      const hasInternalVideo = links.some((a) => {
        const href = a.getAttribute('href') || '';
        return isSameSite(href) && /\/video\//i.test(href);
      });
      if (hasExternal && !hasInternalVideo) {
        // 侧栏/页脚合作方网格
        if (
          el.matches?.('a[href]') ||
          el.querySelector?.('img[src*="partners"]') ||
          /partner|sponsor|affiliate/i.test(el.className || '') ||
          /theporndude|undress|leakify|pornworks|adult zone|telegram/i.test(el.textContent || '')
        ) {
          return true;
        }
      }
    } catch {
      /* ignore */
    }
    return false;
  }

  function killPreroll(root = document) {
    if (!SETTINGS.stripPreroll) return;

      root.querySelectorAll?.('.plyr__ads').forEach((ads) => {

          ads.querySelectorAll('video').forEach((v) => {
            try {
              v.pause();
              v.removeAttribute('src');
              v.load?.();
            } catch {
              /* ignore */
            }
            v.remove();
          });
          ads.innerHTML = '';
          ads.style.display = 'none';
          ads.style.pointerEvents = 'none';
          ads.setAttribute('aria-hidden', 'true');
        } catch {
          /* ignore */
        }
      });

      // 移除广告/追踪 iframe（不误伤空 src 占位）
      root.querySelectorAll?.('iframe').forEach((iframe) => {
        const src = iframe.src || iframe.getAttribute('src') || '';
        if (src && (isBlockedUrl(src) || /imasdk|doubleclick|googlesyndication|tsyndicate/i.test(src))) {
          iframe.remove();
        }
      });
    } catch {
      /* ignore */
    }
  }

  function removeExternalAnchors(root = document) {
    try {
      const scope = root.querySelectorAll ? root : document;
      const nodes = [];
      if (root?.matches?.('a[href]')) nodes.push(root);
      scope.querySelectorAll?.('a[href]').forEach((a) => nodes.push(a));

      for (const anchor of nodes) {
        if (!anchor?.isConnected) continue;
        const href = anchor.getAttribute('href') || '';
        if (!isExternalNav(href) && !isBlockedUrl(href)) continue;

        // 优先删掉整块推广卡片
        const card =
          anchor.closest(
            'a[href], li, [class*="partner" i], [class*="sponsor" i], [data-partner], grid, .grid > div'
          ) || anchor;

        // 避免误删主导航：主导航都是站内路径
        if (anchor.closest?.('nav, aside nav, header nav')) {
          // 侧栏里的外链（Telegram / partners）可删
          if (isExternalNav(href)) {
            const row = anchor.closest('a, div, li, button') || anchor;
            row.remove();
          }
          continue;
        }

        if (card && card !== document.body && card !== document.documentElement) {
          // 若父级是合作方网格中的单项
          const parent = card.parentElement;
          card.remove();
          // 空网格清理
          if (parent && parent !== document.body && parent.children.length === 0) {
            const maybeSection = parent.closest('section, div');
            if (
              maybeSection &&
              maybeSection !== document.body &&
              /partner|sponsor|recommended links|friend/i.test(maybeSection.textContent || '')
            ) {
              // 仅当几乎全是外链文案时删除
            }
          }
        } else {
          anchor.remove();
        }
      }
    } catch {
      /* ignore */
    }
  }

  function patchCreateElement() {
    const nativeCreate = Document.prototype.createElement;
    Document.prototype.createElement = function patchedCreate(tagName, options) {
      const el = nativeCreate.call(this, tagName, options);
      const tag = String(tagName || '').toLowerCase();
      if (tag === 'script' || tag === 'iframe' || tag === 'embed' || tag === 'object') {
        try {
          const nativeSetAttribute = el.setAttribute.bind(el);
          el.setAttribute = function (name, value) {
            if (String(name).toLowerCase() === 'src' && isBlockedUrl(value)) {
              log('block createElement src', tag, String(value).slice(0, 120));
              return;
            }
            return nativeSetAttribute(name, value);
          };
          let currentSrc = '';
          Object.defineProperty(el, 'src', {
            configurable: true,
            enumerable: true,
            get() {
              return currentSrc || el.getAttribute?.('src') || '';
            },
            set(v) {
              const val = String(v || '');
              if (isBlockedUrl(val)) {
                currentSrc = '';
                log('block createElement .src', tag, val.slice(0, 120));
                return;
              }
              currentSrc = val;
              nativeSetAttribute('src', val);
            },
          });
        } catch {
          /* ignore */
        }
      }
      return el;
    };
  }

        .replace(/[\\/:*?"<>|\u0000-\u001F]+/g, '_')

    const tag = label && !/^mp4$/i.test(label) && !/^直链$/i.test(label) ? `_${String(label).replace(/\s+/g, '')}` : '';

  function formatSize(bytes) {
    const n = Number(bytes);
    if (!n || n < 0) return '';
    if (n < 1024) return `${n} B`;
    if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`;
    if (n < 1024 * 1024 * 1024) return `${(n / 1024 / 1024).toFixed(1)} MB`;
    return `${(n / 1024 / 1024 / 1024).toFixed(2)} GB`;
  }

  function addSource(label, url, extra) {
    if (!url || !isMp4Url(url)) return;
    const absU = safeUrl(url);
    if (!absU) return;
    // 只要本站 CDN / 允许媒体
    if (!isMediaAsset(absU.href) && !/(^|\.)cdn\.jerkmotion\.com$/i.test(absU.hostname)) return;
    if (/\/partners\//i.test(absU.pathname)) return;

    const abs = absU.href;
    const key = abs.split('?')[0];
    const item = {
      label: label || 'MP4',
      url: abs,
      server: (extra && extra.server) || '',
      size: (extra && extra.size) || '',
    };
    const idx = state.sources.findIndex((s) => s.url.split('?')[0] === key);
    if (idx >= 0) {
      const old = state.sources[idx];
      if (!old.size && item.size) old.size = item.size;
      if (item.server && !old.server) old.server = item.server;
      if (item.label && (old.label === 'MP4' || old.label === '直链')) old.label = item.label;
    } else {
      state.sources.push(item);
      log('source', item.label, abs.slice(0, 100));
    }
    if (extra && extra.title) state.title = String(extra.title);
    refreshUi();
  }

  function captureFromUrl(url, label) {
    if (!isMp4Url(url)) return;
    // 过滤预览小片
    if (/thumbnails-previews|\/preview/i.test(url) && !/\/videos\//i.test(url)) return;
    if (/\/videos\//i.test(url)) {
      addSource(label || 'MP4', url);
    }
  }

  function captureFromApiPayload(payload) {
    try {
      const data = payload && payload.data ? payload.data : payload;
      if (!data || typeof data !== 'object') return;
      if (data.title) state.title = String(data.title);
      if (data.slug) state.slug = String(data.slug);

      const streams = Array.isArray(data.streams) ? data.streams : [];
      streams.forEach((s, i) => {
        if (!s || !s.stream_url) return;
        if (!s.is_active && s.is_active !== undefined && s.is_active !== 1) return;
        const label =
          s.server_name ||
          s.play_type ||
          (s.priority != null ? `线路${s.priority}` : `线路${i + 1}`);
        addSource(String(label), s.stream_url, { server: s.server_name || '' });
      });

      // 偶发 downloads 字段
      const dls = Array.isArray(data.downloads) ? data.downloads : [];
      dls.forEach((d, i) => {
        const u = d && (d.url || d.download_url || d.stream_url);
        if (u) addSource(d.label || d.quality || `下载${i + 1}`, u);
      });
    } catch (err) {
      log('capture api', err);
    }
  }

  function bindTap(el, handler) {
    if (!el) return;
    let touched = false;
    let lastTs = 0;
    const fire = (e) => {
      const now = Date.now();
      if (now - lastTs < 350) {
        try { e.preventDefault(); e.stopPropagation(); } catch { /* ignore */ }
        return;
      }
      lastTs = now;
      try {
        e.preventDefault();
        e.stopPropagation();
        if (e.stopImmediatePropagation) e.stopImmediatePropagation();
      } catch { /* ignore */ }
      handler(e);
    };

    el.addEventListener(
      'touchend',
      (e) => {
        if (e.touches && e.touches.length > 0) return;
        touched = true;
        fire(e);
        setTimeout(() => { touched = false; }, 450);
      },
      { passive: false }
    );

    el.addEventListener(
      'pointerup',
      (e) => {
        if (e.pointerType === 'mouse' && !IS_TOUCH) return;
        if (touched) return;
        if (e.button != null && e.button !== 0) return;
        fire(e);
      },
      { passive: false }
    );

    el.addEventListener('click', (e) => {
      if (touched) {
        try { e.preventDefault(); e.stopPropagation(); } catch { /* ignore */ }
        return;
      }
      fire(e);
    });
  }

  function setStatus(text, busy) {
    const btn = document.getElementById('jm-dl-btn');
    const status = document.getElementById('jm-dl-status');

      if (btn) btn.dataset.busy = busy ? '1' : '0';

      }

  }

  function anchorDownload(url, filename) {
    const a = document.createElement('a');
    a.href = url;
    a.download = filename || 'video.mp4';
    a.target = '_blank';
    a.rel = 'noopener noreferrer';
    a.style.cssText = 'position:fixed;left:-9999px;top:0;opacity:0;pointer-events:none';
    // 标记为本工具触发，避免被外链守卫误伤（本就同站 CDN）
    a.setAttribute('data-jm-dl', '1');
    document.body.appendChild(a);
    a.click();
    setTimeout(() => {
      try {
        a.remove();
      } catch {
        /* ignore */
      }
    }, 2000);
  }

  async function copyText(text) {
    let ok = false;
    try {
      if (navigator.clipboard && navigator.clipboard.writeText) {
        await navigator.clipboard.writeText(text);
        ok = true;
      }
    } catch { /* ignore */ }

    if (!ok) {
      const ta = document.createElement('textarea');
      ta.value = text;
      if (IS_IOS) {
        ta.contentEditable = 'true';
        ta.readOnly = false;
      } else {
        ta.setAttribute('readonly', '');
      }
      ta.style.cssText =
        'position:fixed;top:10px;left:10px;width:2px;height:2px;padding:0;border:0;opacity:0.01;z-index:2147483647;';
      document.body.appendChild(ta);
      ta.focus();
      ta.setSelectionRange(0, ta.value.length);
      try {
        if (IS_IOS) {
          const range = document.createRange();
          range.selectNodeContents(ta);
          const sel = window.getSelection();
          sel.removeAllRanges();
          sel.addRange(range);
          ta.setSelectionRange(0, ta.value.length);
        }
        ok = document.execCommand('copy');
      } catch {
        ok = false;
      }
      try { ta.remove(); } catch { /* ignore */ }
    }

    if (!ok && IS_MOBILE && navigator.share) {
      try {
        await navigator.share({ title: state.title || 'video', text, url: text });
        return 'share';
      } catch (err) {
        if (err && err.name === 'AbortError') return 'cancel';
      }
    }

    if (!ok) {
      window.prompt(IS_IOS ? '长按全选并复制链接：' : '复制视频链接：', text);
      return 'prompt';
    }
    return 'copy';
  }

  function openMediaUrl(url) {
    if (!url) return false;
    try {
      const win = window.open(url, '_blank');
      if (win) {
        try { win.opener = null; } catch { /* ignore */ }
        return true;
      }
    } catch { /* ignore */ }
    try {
      const a = document.createElement('a');
      a.href = url;
      a.target = '_blank';
      a.rel = 'noopener noreferrer';
      a.setAttribute('data-jm-dl', '1');
      a.style.cssText = 'position:fixed;left:-9999px;top:0;opacity:0';
      document.body.appendChild(a);
      a.click();
      setTimeout(() => { try { a.remove(); } catch { /* ignore */ } }, 1500);
      return true;
    } catch { /* ignore */ }
    if (IS_IOS || IS_MOBILE) {
      try {
        location.href = url;
        return true;
      } catch { /* ignore */ }
    }
    return false;
  }

  function iframeDownload(url) {
    try {
      const iframe = document.createElement('iframe');
      iframe.style.cssText =
        'position:fixed;left:-9999px;top:0;width:1px;height:1px;opacity:0;pointer-events:none;border:0';
      iframe.src = url;
      iframe.setAttribute('data-jm-dl', '1');
      document.body.appendChild(iframe);
      setTimeout(() => { try { iframe.remove(); } catch { /* ignore */ } }, 60000);
      return true;
    } catch {
      return false;
    }
  }

  /**
   * CDN 无 CORS，页面内 fetch 读不了正文。
   * 下载策略：浏览器直链（a[download]）→ 新标签打开 → 复制给下载器。
   * 若运行环境注入了 GM_download / GM_xmlhttpRequest，则优先走油猴通道（可自定义文件名）。
   */
  function getGmDownload() {
    try {
      if (typeof GM_download === 'function') return GM_download;
    } catch {
      /* ignore */
    }
    try {
      if (typeof GM !== 'undefined' && typeof GM.download === 'function') return GM.download.bind(GM);
    } catch {
      /* ignore */
    }
    return null;
  }

  function getGmXhr() {
    try {
      if (typeof GM_xmlhttpRequest === 'function') return GM_xmlhttpRequest;
    } catch {
      /* ignore */
    }
    try {
      if (typeof GM !== 'undefined' && typeof GM.xmlHttpRequest === 'function') {
        return GM.xmlHttpRequest.bind(GM);
      }
    } catch {
      /* ignore */
    }
    return null;
  }

  function downloadViaGm(url, filename) {
    const gmDl = getGmDownload();
    if (gmDl) {
      return new Promise((resolve, reject) => {
        try {
          const ret = gmDl({
            url,
            name: filename,
            saveAs: !IS_MOBILE,
            onload: () => resolve('gm_download'),
            onerror: (e) => reject(e || new Error('GM_download failed')),
            ontimeout: () => reject(new Error('GM_download timeout')),
          });
          // 部分实现返回 Promise
          if (ret && typeof ret.then === 'function') {
            ret.then(() => resolve('gm_download')).catch(reject);
          }
        } catch (err) {
          reject(err);
        }
      });
    }

    const gmXhr = getGmXhr();
    if (!gmXhr) return Promise.reject(new Error('no gm'));

    return new Promise((resolve, reject) => {
      const chunks = [];
      let total = 0;
      let received = 0;
      gmXhr({
        method: 'GET',
        url,
        responseType: 'blob',
        onprogress: (e) => {
          if (!e) return;
          if (e.lengthComputable && e.total) {
            total = e.total;
            received = e.loaded;
            const pct = Math.round((received / total) * 100);
            setStatus(`下载 ${pct}% · ${formatSize(received)}`, true);
          } else if (e.loaded) {
            setStatus(`已接收 ${formatSize(e.loaded)}`, true);
          }
        },
        onload: async (res) => {
          try {
            const blob =
              res.response instanceof Blob
                ? res.response
                : new Blob([res.response], { type: 'video/mp4' });
            // 手机优先系统分享
            if (IS_MOBILE && navigator.canShare) {
              try {
                const file = new File([blob], filename, { type: 'video/mp4' });
                if (navigator.canShare({ files: [file] })) {
                  await navigator.share({ files: [file], title: filename });
                  resolve('share');
                  return;
                }
              } catch (err) {
                if (err && err.name === 'AbortError') {
                  reject(err);
                  return;
                }
              }
            }
            const obj = URL.createObjectURL(blob);
            anchorDownload(obj, filename);
            setTimeout(() => URL.revokeObjectURL(obj), IS_IOS ? 180000 : 60000);
            resolve('gm_xhr_blob');
          } catch (err) {
            reject(err);
          }
        },
        onerror: () => reject(new Error('GM_xhr failed')),
        ontimeout: () => reject(new Error('GM_xhr timeout')),
      });
    });
  }

  async function startDownload(src) {
    const item = src || currentSource();
    if (!item) {
      setStatus('尚未捕获到视频，请先播放或等待加载');
      probeFromPage();
      return;
    }
    if (state.busy) return;

    const filename = buildFilename(item.label);
    state.busy = true;
    state.controller = typeof AbortController !== 'undefined' ? new AbortController() : null;
    setStatus(IS_MOBILE ? `下载 ${item.label}…` : `正在下载 ${item.label}…`, true);
    refreshUi();

    try {
      // 1) 油猴通道（若宿主提供）
      if (getGmDownload() || getGmXhr()) {
        try {
          const mode = await downloadViaGm(item.url, filename);
          setStatus(
            mode === 'share' ? '已分享/保存' : IS_MOBILE ? '已触发下载' : '下载已开始',
            false
          );
          return;
        } catch (err) {
          if (err && err.name === 'AbortError') {
            setStatus('已取消', false);
            return;
          }
          log('gm download fallback', err);
        }
      }

      // 2) iOS / iPadOS：a[download] 基本无效
      if (IS_IOS) {
        const opened = openMediaUrl(item.url);
        if (!opened) anchorDownload(item.url, filename);
        setStatus('若未开始：点「复制/分享」用「文件 / 迅雷」', false);
        return;
      }

      // 3) 安卓/手机：a[download] + 隐藏 iframe（Via/UC/X5 更稳）
      if (IS_ANDROID || IS_MOBILE) {
        anchorDownload(item.url, filename);
        setTimeout(() => { try { iframeDownload(item.url); } catch { /* ignore */ } }, 200);
        setStatus(
          IS_ANDROID
            ? '已触发下载，可看通知栏；不行就复制/分享直链'
            : '已触发下载；不行就复制/分享直链',
          false
        );
        return;
      }

      // 4) 桌面
      anchorDownload(item.url, filename);
      setStatus('已触发浏览器下载（跨域时文件名可能是 CDN 原名）', false);
    } catch (e) {
      setStatus('下载失败，请复制直链', false);
      log('download err', e);
    } finally {
      state.busy = false;
      state.controller = null;
      refreshUi();
    }
  }

          const bottomGap = Math.max(0, window.innerHeight - vv.height - vv.offsetTop);
          root.style.bottom =
            'max(12px, calc(56px + env(safe-area-inset-bottom, 0px) + ' + bottomGap + 'px))';
        } catch { /* ignore */ }
      };
      window.visualViewport.addEventListener('resize', syncViewport);
      window.visualViewport.addEventListener('scroll', syncViewport);
      syncViewport();
    }

    const closeMenu = (e) => {
      if (!menu.hidden && !root.contains(e.target)) menu.hidden = true;
    };
    document.addEventListener('click', closeMenu, true);
    document.addEventListener('touchstart', closeMenu, { passive: true, capture: true });

    refreshUi();
  }

  function installSpaWatch() {
    let last = location.href;
    const onChange = () => {
      if (location.href === last) return;
      last = location.href;
      resetDownloadState(true);
      // 等 Next 路由渲染
      setTimeout(() => {
        cleanAds(document);
        if (isDetailPage()) {
          probeFromPage();
          createFloatingUI();
        } else {
          const root = document.getElementById(UI_ID);
          if (root) root.remove();
        }
      }, 300);
      setTimeout(() => {
        if (isDetailPage()) {
          probeFromPage();
          createFloatingUI();
          refreshUi();
        }
      }, 1200);
    };

    try {
      const wrap = (type) => {
        const raw = history[type];
        if (typeof raw !== 'function') return;
        history[type] = function patchedHistory() {
          const ret = raw.apply(this, arguments);
          try {
            onChange();
          } catch {
            /* ignore */
          }
          return ret;
        };
      };
      wrap('pushState');
      wrap('replaceState');
    } catch {
      /* ignore */
    }
    window.addEventListener('popstate', onChange);
    // 兜底轮询（部分路由不走 history 钩子）
    setInterval(() => {
      if (location.href !== last) onChange();
    }, 1000);
  }

  function installMediaHooks() {
    // 播放器 src 变化时捕获
    try {
      const desc = Object.getOwnPropertyDescriptor(HTMLMediaElement.prototype, 'src');
      if (desc && desc.set) {
        Object.defineProperty(HTMLMediaElement.prototype, 'src', {
          configurable: true,
          enumerable: desc.enumerable,
          get: desc.get,
          set(v) {
            try {
              const s = String(v || '');
              if (isMp4Url(s)) captureFromUrl(s, '播放中');
            } catch {
              /* ignore */
            }
            return desc.set.call(this, v);
          },
        });
      }
    } catch {
      /* ignore */
    }

    document.addEventListener(
      'loadedmetadata',
      (e) => {
        try {
          const v = e.target;
          if (!v || v.tagName !== 'VIDEO') return;
          if (String(v.getAttribute('title') || '').toLowerCase() === 'advertisement') return;
          const src = v.currentSrc || v.src;
          if (src) captureFromUrl(src, '播放中');
        } catch {
          /* ignore */
        }
      },
      true
    );
  }

    function bootUi() {
    try { cleanAds(document); } catch { /* ignore */ }
    probeFromPage();
    createFloatingUI();
    ensureContentPlayable();
  }

  function scheduleMobileRetries() {
    UI_RETRY_MS.forEach((ms) => {
      setTimeout(() => {
        try {
          cleanAds(document);
          ensureContentPlayable();
          if (!isDetailPage()) return;
          captureFromDom();
          probeFromPage();
          createFloatingUI();
          refreshUi();
        } catch (err) {
          log('retry', err);
        }
      }, ms);
    });
  }

  // ---------- 启动 ----------
  function boot() {
    log('boot', {
      href: location.href,
      mobile: IS_MOBILE,
      ios: IS_IOS,
      android: IS_ANDROID,
      touch: IS_TOUCH,
    });
    hardenPrivacy();
    installImaStub();
    patchCreateElement();
    patchElementSetters();
    installNetworkGuards();
    installNavigationGuard();
    installMediaHooks();
    installAdCleaner();
    installSpaWatch();

    const startUi = () => {
      bootUi();
      scheduleMobileRetries();
    };
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', startUi, { once: true });
    } else {
      startUi();
    }
    window.addEventListener('load', () => bootUi(), { once: true });
    window.addEventListener('pageshow', (ev) => {
      if (ev.persisted || IS_MOBILE) {
        cleanAds(document);
        bootUi();
      }
    });
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'visible') {
        cleanAds(document);
        ensureContentPlayable();
        if (isDetailPage()) {
          captureFromDom();
          createFloatingUI();
        }
      }
    });
    window.addEventListener('orientationchange', () => {
      setTimeout(() => {
        createFloatingUI();
        ensureContentPlayable();
      }, 300);
    });
  }

  boot();

}

    if (window[BOOT_FLAG]) return;
  } catch (e) {}

        console.error('[jerkmotion-clean] boot failed', err);
      }
    }
    return;
  }

  try {
    JM_MAIN();
  } catch (err) {
    console.error('[jerkmotion-clean] main failed', err);
  }
})();
