  function replaceValue(target, name, value) {
    try {
      Object.defineProperty(target, name, {
        configurable: true,
        enumerable: true,
        writable: true,
        value,
      });
    } catch {
      try { target[name] = value; } catch { /* ignore */ }
    }
  }

  function blockMethod(target, name, label) {
    if (!target || typeof target[name] !== 'function') return;

  function safeUrl(value) {
    try {
      return new URL(String(value), location.href);
    } catch {
      return null;
    }
  }

  function hostOf(value) {
    const url = safeUrl(value);
    return url ? url.hostname : '';
  }

  function isSameSite(value) {
    const host = hostOf(value);
    if (!host) return true;
    if (SITE_HOST_RE.test(host)) return true;
    if (host === location.hostname) return true;
    return false;
  }

  function isMediaHost(value) {
    const host = hostOf(value);
    return !host || MEDIA_HOST_RE.test(host);
  }

  function isTracker(value) {
    return TRACKER_RE.test(String(value || ''));
  }

  function isSiteTelemetry(value) {
    return SITE_TELEMETRY_RE.test(String(value || ''));
  }

  function isAdHost(value) {
    return AD_HOST_HINT_RE.test(String(value || ''));
  }

  function isExternalDanger(value) {
    const raw = String(value || '').trim();
    if (!raw || raw.startsWith('#') || raw.toLowerCase().startsWith('javascript:')) return false;
    if (isSameSite(raw)) return false;
    if (isMediaHost(raw)) return false;
    if (/^(?:blob:|data:)/i.test(raw)) return false;
    return true;
  }

  function isAdLikeAnchor(anchor) {

    const href = anchor.getAttribute('href') || '';
    return !!(

      || anchor.hasAttribute('data-ad_slot_key')

    );

  function shouldBlockNetwork(url) {
    if (!url) return false;
    if (isTracker(url)) return true;
    if (isSiteTelemetry(url)) return true;
    if (isAdHost(url) && !isMediaHost(url) && !isSameSite(url)) return true;
    return false;
  }

  function shouldBlockScriptSrc(val) {
    const src = String(val || '');
    if (!src) return false;
    if (shouldBlockNetwork(src)) return true;
    if (/\/static\/frontend\/js\/web_tracker/i.test(src)) return true;
    if (/\/static\/frontend\/js\/web-sdk/i.test(src)) return true;
    if (!isSameSite(src) && !/^\/(?!\/)/.test(src) && !/^(?:blob:|data:)/i.test(src)) return true;
    return false;
  }

  function installNetworkGuards() {
    if (!SETTINGS.blockTrackers) return;

    const nativeFetch = window.fetch?.bind(window);
    if (nativeFetch) {
      window.fetch = function patchedFetch(input, init) {
        const url = typeof input === 'string' || input instanceof URL
          ? String(input)
          : String(input?.url || '');
        if (shouldBlockNetwork(url)) {
          blockedNet += 1;
          log('block fetch', url);
          return Promise.reject(securityError(`fetch ${url}`));
        }
        return nativeFetch(input, init);
      };
    }

    const nativeOpen = XMLHttpRequest.prototype.open;
    const nativeSend = XMLHttpRequest.prototype.send;
    XMLHttpRequest.prototype.open = function patchedOpen(method, url, ...rest) {
      this.__mh18url = String(url || '');
      if (shouldBlockNetwork(this.__mh18url)) {
        this.__mh18block = true;
        blockedNet += 1;
        log('block xhr', this.__mh18url);
        return undefined;
      }
      return nativeOpen.call(this, method, url, ...rest);
    };
    XMLHttpRequest.prototype.send = function patchedSend(body) {
      if (this.__mh18block) {
        try { this.abort(); } catch { /* ignore */ }
        return undefined;
      }
      return nativeSend.call(this, body);
    };

    const nativeCreate = Document.prototype.createElement;
    Document.prototype.createElement = function patchedCreate(tagName, options) {
      const el = nativeCreate.call(this, tagName, options);
      const tag = String(tagName || '').toLowerCase();
      if (tag === 'script' || tag === 'iframe' || tag === 'embed' || tag === 'object') {
        const nativeSetAttribute = el.setAttribute.bind(el);
        el.setAttribute = function patchedSetAttribute(name, value) {
          if (String(name).toLowerCase() === 'src') {
            const val = String(value || '');
            if (tag === 'script' && shouldBlockScriptSrc(val)) {
              blockedNet += 1;
              log('block createElement script', val);
              return undefined;
            }
            if (tag !== 'script' && (shouldBlockNetwork(val) || (isExternalDanger(val) && !isMediaHost(val)))) {
              blockedNet += 1;
              log('block createElement src', tag, val);
              return undefined;
            }
          }
          return nativeSetAttribute(name, value);
        };

        try {
          let currentSrc = el.src || '';
          Object.defineProperty(el, 'src', {
            configurable: true,
            get() { return currentSrc; },
            set(v) {
              const val = String(v || '');
              if (tag === 'script' && shouldBlockScriptSrc(val)) {
                currentSrc = '';
                blockedNet += 1;
                return;
              }
              if (tag !== 'script' && val && (shouldBlockNetwork(val) || (isExternalDanger(val) && !isMediaHost(val)))) {
                currentSrc = '';
                blockedNet += 1;
                return;
              }
              currentSrc = val;
              nativeSetAttribute('src', val);
            },
          });
        } catch { /* ignore */ }
      }
      return el;
    };

    const neutralizeTrackerNode = (node) => {
      if (!node || node.nodeType !== Node.ELEMENT_NODE) return;
      const tag = node.tagName;
      if (tag === 'SCRIPT') {
        const src = node.getAttribute('src') || node.src || '';
        const text = node.textContent || '';
        const looksTrackerText = /gtag\(|ym\(|Yandex\.Metrika|dataLayer\.push/i.test(text)
          && /G-VVD7MLW33L|105105014|metrika/i.test(text + src);
        if (shouldBlockScriptSrc(src) || looksTrackerText) {
          try {
            node.type = 'text/plain';
            node.removeAttribute('src');
            node.textContent = '/* blocked by 18mh-clean */';
            node.remove();
            blockedNet += 1;
          } catch { /* ignore */ }
        }
        return;
      }
      if (tag === 'IFRAME' || tag === 'EMBED' || tag === 'OBJECT') {
        const src = node.getAttribute('src') || node.src || node.getAttribute('data') || '';
        if (shouldBlockNetwork(src) || (isExternalDanger(src) && !isMediaHost(src))) {
          try { node.remove(); blockedNet += 1; } catch { /* ignore */ }
        }
      }
    };

    const killExistingTrackers = (root = document) => {
      root.querySelectorAll?.('script, iframe[src], embed[src], object[data]').forEach(neutralizeTrackerNode);
    };

    const trackerObs = new MutationObserver((records) => {
      for (const record of records) {
        record.addedNodes.forEach((node) => {
          neutralizeTrackerNode(node);
          if (node?.querySelectorAll) killExistingTrackers(node);
        });
      }
    });

    const startTrackerObs = () => {
      const root = document.documentElement || document;
      try { trackerObs.observe(root, { childList: true, subtree: true }); } catch { /* ignore */ }
      killExistingTrackers();
    };
    startTrackerObs();
  }

  /* ---------------- 导航守卫（手机 touch 关键） ---------------- */

  function installNavigationGuard() {
    if (!SETTINGS.blockExternalNav) return;

    const nativeOpen = window.open?.bind(window);
    if (nativeOpen) {
      window.open = function patchedOpen(url, ...rest) {
        if (url && isExternalDanger(url)) {
          log('block window.open', url);
          return null;
        }
        // 手机上无第三个参数的 open 也常被广告用来跳转
        return nativeOpen(url, ...rest);
      };
    }

    // 部分壳浏览器 location.assign 跳广告
    try {
      const loc = window.location;
      const nativeAssign = loc.assign?.bind(loc);
      const nativeReplace = loc.replace?.bind(loc);
      if (nativeAssign) {
        loc.assign = function patchedAssign(url) {
          if (isExternalDanger(url) || isAdHost(url)) {
            log('block location.assign', url);
            return;
          }
          return nativeAssign(url);
        };
      }
      if (nativeReplace) {
        loc.replace = function patchedReplace(url) {
          if (isExternalDanger(url) || isAdHost(url)) {
            log('block location.replace', url);
            return;
          }
          return nativeReplace(url);
        };
      }
    } catch { /* 部分 WebView 禁止改 location 方法 */ }

    const blockExternalNav = (event) => {
      const path = typeof event.composedPath === 'function' ? event.composedPath() : null;
      let anchor = event.target?.closest?.('a[href]');
      if (!anchor && path) {
        anchor = path.find((n) => n && n.tagName === 'A' && n.getAttribute?.('href'));
      }
      if (!anchor) return;

      const href = anchor.getAttribute('href') || '';
      if (!href || href.startsWith('#') || href.toLowerCase().startsWith('javascript:')) return;

      if (isAdLikeAnchor(anchor) || isExternalDanger(href)) {
        // 手机必须 cancelable + non-passive 才能拦住默认跳转
        if (event.cancelable) event.preventDefault();
        event.stopPropagation();
        try { event.stopImmediatePropagation(); } catch { /* ignore */ }
        log('block nav', href, event.type);
      }
    };

    // passive:false —— Chrome/Android 默认 touch* 为 passive，否则 preventDefault 无效

  function isProtectedNode(el) {
    if (!el || el.nodeType !== Node.ELEMENT_NODE) return true;
    if (el.id === 'login_modal' || el.id === UI_BADGE_ID) return true;
    if (el.getAttribute?.('data-mh18') === 'badge') return true;

  function removeNode(el) {
    if (!el || !el.remove || isProtectedNode(el)) return false;
    try {
      el.remove();
      removedCount += 1;
      return true;

        el.style.setProperty('display', 'none', 'important');
        el.style.setProperty('pointer-events', 'none', 'important');
        el.style.setProperty('visibility', 'hidden', 'important');
        removedCount += 1;
        return true;
      } catch {
        return false;
      }
    }
  }

  const CRC32_TABLE = Object.freeze(Array.from({ length: 256 }, (_, value) => {
    let crc = value;
    for (let bit = 0; bit < BITS_PER_BYTE; bit += 1) {
      crc = (crc & 1) ? (CRC32_POLYNOMIAL ^ (crc >>> 1)) : (crc >>> 1);
    }
    return crc >>> 0;
  }));

  function resolveDownloadUrl(value, base = location.href) {
    let url;
    try {
      url = new URL(String(value || ''), base);
    } catch {
      throw new Error('资源地址无效');
    }
    if (url.protocol !== 'https:') throw new Error(`拒绝非 HTTPS 资源：${url.protocol}`);
    if (!DOWNLOAD_HOST_RE.test(url.hostname)) throw new Error(`拒绝未知资源域名：${url.hostname}`);
    return url;
  }

  async function fetchDownloadResponse(value, { signal } = {}) {
    const url = resolveDownloadUrl(value);
    const credentials = SITE_HOST_RE.test(url.hostname) ? 'include' : 'omit';
    let response;
    try {
      response = await fetch(url.href, { signal, credentials });
    } catch (error) {
      if (signal?.aborted) throw error;
      throw new Error(`资源请求失败：${url.hostname}${url.pathname}（${error?.message || String(error)}）`, {
        cause: error,
      });
    }
    if (!response.ok) throw new Error(`资源请求失败：HTTP ${response.status} ${url.pathname}`);
    return response;
  }

  async function fetchDownloadBytes(value, signal) {
    const response = await fetchDownloadResponse(value, { signal });
    return new Uint8Array(await response.arrayBuffer());
  }

  async function fetchDownloadText(value, signal) {
    const response = await fetchDownloadResponse(value, { signal });
    return response.text();
  }

  async function mapDownloadItems(items, options) {
    const results = new Array(items.length);
    let nextIndex = 0;
    let completed = 0;
    async function runWorker() {
      while (nextIndex < items.length) {
        if (options.signal?.aborted) throw options.signal.reason || new DOMException('已取消', 'AbortError');
        const index = nextIndex;
        nextIndex += 1;
        results[index] = await options.worker(items[index], index);
        completed += 1;
        options.onProgress?.(completed, items.length);
      }
    }
    const count = Math.min(DOWNLOAD_CONCURRENCY, items.length);
    await Promise.all(Array.from({ length: count }, runWorker));
    return results;
  }

  function crc32(bytes) {
    let crc = CRC32_INITIAL;
    for (const byte of bytes) crc = CRC32_TABLE[(crc ^ byte) & 0xff] ^ (crc >>> 8);
    return (crc ^ CRC32_INITIAL) >>> 0;
  }

  function zipTimestamp(date = new Date()) {
    const year = Math.max(ZIP_MIN_YEAR, date.getFullYear());
    return Object.freeze({
      time: (date.getHours() << 11) | (date.getMinutes() << 5) | (date.getSeconds() >>> 1),
      date: ((year - ZIP_MIN_YEAR) << 9) | ((date.getMonth() + 1) << 5) | date.getDate(),
    });
  }

  function createLocalZipHeader(meta) {
    const header = new Uint8Array(ZIP_LOCAL_HEADER_BYTES);
    const view = new DataView(header.buffer);
    view.setUint32(0, ZIP_LOCAL_SIGNATURE, true);
    view.setUint16(4, ZIP_VERSION, true);
    view.setUint16(6, ZIP_UTF8_FLAG, true);
    view.setUint16(10, meta.stamp.time, true);
    view.setUint16(12, meta.stamp.date, true);
    view.setUint32(14, meta.crc, true);
    view.setUint32(18, meta.size, true);
    view.setUint32(22, meta.size, true);
    view.setUint16(26, meta.name.length, true);
    return header;
  }

  function createCentralZipHeader(meta) {
    const header = new Uint8Array(ZIP_CENTRAL_HEADER_BYTES);
    const view = new DataView(header.buffer);
    view.setUint32(0, ZIP_CENTRAL_SIGNATURE, true);
    view.setUint16(4, ZIP_VERSION, true);
    view.setUint16(6, ZIP_VERSION, true);
    view.setUint16(8, ZIP_UTF8_FLAG, true);
    view.setUint16(12, meta.stamp.time, true);
    view.setUint16(14, meta.stamp.date, true);
    view.setUint32(16, meta.crc, true);
    view.setUint32(20, meta.size, true);
    view.setUint32(24, meta.size, true);
    view.setUint16(28, meta.name.length, true);
    view.setUint32(42, meta.offset, true);
    return header;
  }

  function createZipEnd(count, centralSize, centralOffset) {
    const end = new Uint8Array(ZIP_END_BYTES);
    const view = new DataView(end.buffer);
    view.setUint32(0, ZIP_END_SIGNATURE, true);
    view.setUint16(8, count, true);
    view.setUint16(10, count, true);
    view.setUint32(12, centralSize, true);
    view.setUint32(16, centralOffset, true);
    return end;
  }

  function createZipArchive(entries) {
    if (entries.length > ZIP32_MAX_ENTRIES) throw new Error('图片数量超过 ZIP32 格式范围');
    const encoder = new TextEncoder();
    const stamp = zipTimestamp();
    const localParts = [];
    const centralParts = [];
    let offset = 0;
    for (const entry of entries) {
      const name = encoder.encode(entry.name);
      const size = entry.data.byteLength;
      if (size > ZIP32_MAX_VALUE) throw new Error(`单个资源过大：${entry.name}`);
      const meta = Object.freeze({ name, size, stamp, offset, crc: crc32(entry.data) });
      const local = createLocalZipHeader(meta);
      localParts.push(local, name, entry.data);
      centralParts.push(createCentralZipHeader(meta), name);
      offset += local.byteLength + name.byteLength + size;
    }
    const centralSize = centralParts.reduce((total, part) => total + part.byteLength, 0);
    if (offset + centralSize > ZIP32_MAX_VALUE) throw new Error('章节资源超过 ZIP32 格式范围');
    return new Blob([...localParts, ...centralParts, createZipEnd(entries.length, centralSize, offset)], {
      type: 'application/zip',
    });
  }

  function saveDownloadBlob(blob, fileName) {

  function readPageInfo(attribute) {

  function bytesMatch(bytes, signature, offset = 0) {
    if (bytes.length < offset + signature.length) return false;
    return signature.every((byte, index) => bytes[offset + index] === byte);
  }

  function detectImageExtension(bytes) {
    if (bytesMatch(bytes, JPEG_SIGNATURE)) return 'jpg';
    if (bytesMatch(bytes, PNG_SIGNATURE)) return 'png';
    if (bytesMatch(bytes, GIF87A_SIGNATURE) || bytesMatch(bytes, GIF89A_SIGNATURE)) return 'gif';
    if (bytesMatch(bytes, RIFF_SIGNATURE)
      && bytesMatch(bytes, WEBP_SIGNATURE, WEBP_FORMAT_OFFSET)) return 'webp';
    const isIsoImage = bytesMatch(bytes, FTYP_SIGNATURE, ISO_BOX_TYPE_OFFSET);
    const isAvif = bytesMatch(bytes, AVIF_SIGNATURE, ISO_BRAND_OFFSET)
      || bytesMatch(bytes, AVIS_SIGNATURE, ISO_BRAND_OFFSET);
    if (isIsoImage && isAvif) return 'avif';
    throw new Error('解密后的图片格式无法识别');
  }

  function throwIfDownloadAborted(signal) {
    if (!signal?.aborted) return;
    throw signal.reason || new DOMException('已取消', 'AbortError');
  }

  function loadComicImageDecryptor() {
    const requireModule = window.requirejs || window.require;
    if (typeof requireModule !== 'function') {
      return Promise.reject(new Error('页面图片解密模块加载器不可用'));
    }
    return new Promise((resolve, reject) => {
      const onLoad = (module) => {
        if (typeof module?.decrypto_image !== 'function') {
          reject(new Error('页面图片解密模块接口不可用'));
          return;
        }
        resolve(module.decrypto_image.bind(module));
      };
      const onError = (error) => reject(new Error(`页面图片解密模块加载失败：${error?.message || String(error)}`, {
        cause: error,
      }));
      try {
        requireModule(['plugins/lazyload'], onLoad, onError);
      } catch (error) {
        onError(error);
      }
    });
  }

  async function fetchDecryptedComicImage(value, options) {
    const url = resolveDownloadUrl(value);
    throwIfDownloadAborted(options.signal);
    let objectUrl;
    try {
      objectUrl = await options.decryptImage(url.href);
      throwIfDownloadAborted(options.signal);
      if (typeof objectUrl !== 'string' || !objectUrl.startsWith('blob:')) {
        throw new Error('页面解密器未返回 Blob 地址');
      }
      const response = await fetch(objectUrl, { signal: options.signal });
      if (!response.ok) throw new Error(`解密图片读取失败：HTTP ${response.status}`);
      const bytes = new Uint8Array(await response.arrayBuffer());
      return Object.freeze({ bytes, extension: detectImageExtension(bytes) });
    } catch (error) {
      throwIfDownloadAborted(options.signal);
      throw new Error(`漫画原图解密失败：${url.pathname}（${error?.message || String(error)}）`, { cause: error });
    } finally {
      if (typeof objectUrl === 'string' && objectUrl.startsWith('blob:')) {
        URL.revokeObjectURL(objectUrl);
      }
    }
  }

  function collectComicImageUrls() {
    const root = document.querySelector('main[data-comic-info] > section');
    if (!root) throw new Error('当前页面不是可下载的漫画章节');
    const urls = [...root.querySelectorAll('img[data-src]')].map((image) => {
      const url = resolveDownloadUrl(image.getAttribute('data-src'));
      if (!/\/upload(?:_|\/)/i.test(url.pathname)) throw new Error(`发现非正文图片：${url.pathname}`);
      return url.href;
    });
    const uniqueUrls = [...new Set(urls)];
    if (!uniqueUrls.length) throw new Error('当前章节没有可下载的原图');
    return uniqueUrls;
  }

  function parseHlsAttributes(line) {
    const attributes = {};
    const body = line.slice(line.indexOf(':') + 1);
    const matcher = /([A-Z0-9-]+)=("[^"]*"|[^,]*)/gi;
    for (const match of body.matchAll(matcher)) {
      const raw = match[2];
      attributes[match[1].toUpperCase()] = raw.startsWith('"') ? raw.slice(1, -1) : raw;
    }
    return Object.freeze(attributes);
  }

  function selectHlsVariant(text, baseUrl) {
    const lines = text.replace(/\r/g, '').split('\n');
    let selected = null;
    for (let index = 0; index < lines.length; index += 1) {
      if (!lines[index].startsWith('#EXT-X-STREAM-INF:')) continue;
      const bandwidth = Number(parseHlsAttributes(lines[index]).BANDWIDTH || 0);
      if (!Number.isSafeInteger(bandwidth) || bandwidth < 0) throw new Error('HLS 码率字段无效');
      const resource = lines.slice(index + 1).find((line) => line.trim() && !line.startsWith('#'));
      if (!resource) throw new Error('HLS 主清单缺少码率地址');
      const candidate = Object.freeze({ bandwidth, url: resolveDownloadUrl(resource.trim(), baseUrl).href });
      if (!selected || candidate.bandwidth > selected.bandwidth) selected = candidate;
    }
    return selected?.url || '';
  }

  function parseHlsKey(line, baseUrl) {
    const attributes = parseHlsAttributes(line);
    if (attributes.METHOD === 'NONE') return null;
    if (attributes.METHOD !== 'AES-128') throw new Error(`不支持的 HLS 加密：${attributes.METHOD || '未知'}`);
    if (!attributes.URI) throw new Error('HLS 密钥地址缺失');
    return Object.freeze({
      url: resolveDownloadUrl(attributes.URI, baseUrl).href,
      iv: attributes.IV || '',
    });
  }

  function parseHlsSegments(text, baseUrl) {
    const lines = text.replace(/\r/g, '').split('\n').map((line) => line.trim()).filter(Boolean);
    if (lines.some((line) => /^#EXT-X-(?:BYTERANGE|MAP):/.test(line))) {
      throw new Error('当前 HLS 使用了尚不支持的分段格式');
    }
    const sequenceLine = lines.find((line) => line.startsWith('#EXT-X-MEDIA-SEQUENCE:'));
    let sequence = Number(sequenceLine?.split(':')[1] || 0);
    if (!Number.isSafeInteger(sequence) || sequence < 0) throw new Error('HLS 媒体序号无效');
    let key = null;
    const segments = [];
    for (const line of lines) {
      if (line.startsWith('#EXT-X-KEY:')) {
        key = parseHlsKey(line, baseUrl);
      } else if (!line.startsWith('#')) {
        segments.push(Object.freeze({ url: resolveDownloadUrl(line, baseUrl).href, key, sequence }));
        sequence += 1;
      }
    }
    if (!segments.length) throw new Error('HLS 清单中没有媒体分段');
    return segments;
  }

  async function loadHlsSegments(initialUrl, signal) {
    const visited = new Set();
    let manifestUrl = resolveDownloadUrl(initialUrl).href;
    while (!visited.has(manifestUrl)) {
      visited.add(manifestUrl);
      const text = await fetchDownloadText(manifestUrl, signal);
      if (!text.trimStart().startsWith('#EXTM3U')) throw new Error('视频清单不是有效的 HLS 文本');
      const variant = selectHlsVariant(text, manifestUrl);
      if (!variant) return parseHlsSegments(text, manifestUrl);
      manifestUrl = variant;
    }
    throw new Error('HLS 清单存在循环引用');
  }

  function readVideoManifestUrl() {
    for (const script of document.scripts) {
      const source = String(script.textContent || '');
      const marker = source.indexOf('const _detail_');
      if (marker < 0) continue;
      const match = source.slice(marker).match(/"url"\s*:\s*"((?:\\.|[^"])*?\.m3u8(?:\\.|[^"])*)"/);
      if (!match) continue;
      return resolveDownloadUrl(JSON.parse(`"${match[1]}"`)).href;
    }
    const mediaUrls = [...document.querySelectorAll('#mse[data-url], video[src], source[src]')]
      .map((element) => element.dataset?.url || element.getAttribute('src') || '')
      .filter((value) => /\.m3u8(?:$|\?)/i.test(value));
    const directUrl = mediaUrls.find((value) => !/\.v1\.m3u8(?:$|\?)/i.test(value));
    if (directUrl) return resolveDownloadUrl(directUrl).href;
    if (mediaUrls.length) throw new Error('页面仅提供防劫持视频清单，无法直接下载');
    throw new Error('页面中没有可用的视频清单');
  }

  function createSegmentIv(value, sequence) {
    if (value) {
      const hex = value.replace(/^0x/i, '');
      if (hex.length !== AES128_IV_HEX_LENGTH || !/^[0-9a-f]+$/i.test(hex)) {
        throw new Error('HLS 分段 IV 无效');
      }
      return Uint8Array.from(hex.match(/.{2}/g), (byte) => Number.parseInt(byte, 16));
    }
    const iv = new Uint8Array(AES128_KEY_BYTES);
    let remaining = BigInt(sequence);
    for (let index = AES128_KEY_BYTES - 1; index >= 0; index -= 1) {
      iv[index] = Number(remaining & 0xffn);
      remaining >>= BigInt(BITS_PER_BYTE);
    }
    return iv;
  }

  function loadHlsKey(keyInfo, cache, signal) {
    if (cache.has(keyInfo.url)) return cache.get(keyInfo.url);
    const pending = fetchDownloadBytes(keyInfo.url, signal).then((bytes) => {
      if (bytes.byteLength !== AES128_KEY_BYTES) throw new Error('HLS AES-128 密钥长度无效');
      return crypto.subtle.importKey('raw', bytes, { name: 'AES-CBC' }, false, ['decrypt']);
    });
    cache.set(keyInfo.url, pending);
    return pending;
  }

  async function decodeHlsSegment(segment, options) {
    const bytes = await fetchDownloadBytes(segment.url, options.signal);
    if (!segment.key) return validateMpegTs(bytes, segment.sequence);
    if (!crypto.subtle) throw new Error('当前浏览器不支持 Web Crypto 视频解密');
    const key = await loadHlsKey(segment.key, options.keyCache, options.signal);
    const iv = createSegmentIv(segment.key.iv, segment.sequence);
    const decrypted = await crypto.subtle.decrypt({ name: 'AES-CBC', iv }, key, bytes);
    return validateMpegTs(new Uint8Array(decrypted), segment.sequence);
  }

  function validateMpegTs(bytes, sequence) {
    if (!bytes.byteLength || bytes.byteLength % MPEG_TS_PACKET_BYTES !== 0) {
      throw new Error(`HLS 分段 ${sequence} 的 MPEG-TS 长度无效`);
    }
    for (let offset = 0; offset < bytes.byteLength; offset += MPEG_TS_PACKET_BYTES) {
      if (bytes[offset] !== MPEG_TS_SYNC_BYTE) {
        throw new Error(`HLS 分段 ${sequence} 的 MPEG-TS 同步字节无效`);
      }
    }
    return bytes;
  }

  async function createOpfsVideoWriter(fileName) {
    if (!navigator.storage?.getDirectory) throw new Error('当前浏览器不支持视频流式写盘');
    const root = await navigator.storage.getDirectory();
    const tempName = `${fileName}.part-${Date.now()}`;
    const handle = await root.getFileHandle(tempName, { create: true });
    const stream = await handle.createWritable();
    return Object.freeze({
      write: (chunk) => stream.write(chunk),
      async finish() {
        await stream.close();
        saveDownloadBlob(await handle.getFile(), fileName);
        setTimeout(() => root.removeEntry(tempName).catch((error) => {
          console.error(LOG_PREFIX, '清理视频临时文件失败', error);
        }), OBJECT_URL_TTL_MS);
      },
      async abort() {
        try { await stream.abort(); } finally { await root.removeEntry(tempName); }
      },
    });
  }

  async function createVideoWriter(fileName) {
    if (typeof window.showSaveFilePicker !== 'function') return createOpfsVideoWriter(fileName);
    const handle = await window.showSaveFilePicker({
      suggestedName: fileName,
      types: [{ description: 'MPEG-TS 视频', accept: { 'video/mp2t': ['.ts'] } }],
    });
    const stream = await handle.createWritable();
    return Object.freeze({
      write: (chunk) => stream.write(chunk),
      finish: () => stream.close(),
      abort: () => stream.abort(),
    });
  }

  function formatBytes(value) {
    if (value < MEBIBYTE) return `${(value / KIBIBYTE).toFixed(1)} KB`;
    if (value < GIBIBYTE) return `${(value / MEBIBYTE).toFixed(1)} MB`;
    return `${(value / GIBIBYTE).toFixed(2)} GB`;
  }

  async function abortVideoWriter(writer, downloadError) {
    try {
      await writer.abort();
    } catch (cleanupError) {
      throw new AggregateError([downloadError, cleanupError], '视频下载失败，且临时文件清理失败');
    }
    throw downloadError;
  }

  async function downloadCurrentVideo({ signal, update }) {
    const videoId = location.pathname.match(/\/mv\/detail\/(\d+)/)?.[1] || 'video';
    const writer = await createVideoWriter(`18mh-video-${videoId}.ts`);
    try {
      update(VIDEO_MANIFEST_PROGRESS, '正在读取播放列表');
      const segments = await loadHlsSegments(readVideoManifestUrl(), signal);
      const keyCache = new Map();
      let written = 0;
      for (let start = 0; start < segments.length; start += DOWNLOAD_CONCURRENCY) {
        const group = segments.slice(start, start + DOWNLOAD_CONCURRENCY);
        const chunks = await Promise.all(group.map((segment) => decodeHlsSegment(segment, { keyCache, signal })));
        for (const chunk of chunks) {
          await writer.write(chunk);
          written += chunk.byteLength;
        }
        const done = Math.min(start + group.length, segments.length);
        update(done / segments.length, `分段 ${done}/${segments.length} · ${formatBytes(written)}`);
      }
      await writer.finish();
    } catch (error) {
      await abortVideoWriter(writer, error);
    }
  }

    if (/\/comic\/chapter\/\d+\/\d+/.test(location.pathname)) {

    if (/\/mv\/detail\/\d+/.test(location.pathname)) {
      return Object.freeze({ label: '下载当前视频 TS', run: downloadCurrentVideo });
    }
    return null;
  }

  function createProgressUpdater(elements) {
    return (ratio, message) => {
      const percent = Math.round(Math.max(0, Math.min(1, ratio)) * 100);
      elements.status.textContent = message;
      elements.progress.style.width = `${percent}%`;
      elements.track.setAttribute('aria-valuenow', String(percent));
    };
  }

  function bindDownloadAction(action, elements) {
    let controller = null;
    const update = createProgressUpdater(elements);
    elements.actionButton.addEventListener('click', async () => {
      if (controller) {
        controller.abort(new DOMException('用户取消下载', 'AbortError'));
        elements.actionButton.textContent = '正在取消';
        return;
      }
      controller = new AbortController();
      elements.actionButton.textContent = '取消下载';
      update(0, '正在准备');
      try {
        await action.run({ signal: controller.signal, update });
        update(1, '下载完成');
      } catch (error) {
        const cancelled = controller.signal.aborted || error?.name === 'AbortError';
        update(0, cancelled ? '已取消' : `失败：${error?.message || String(error)}`);
        if (!cancelled) console.error(LOG_PREFIX, '资源下载失败', error);
      } finally {
        controller = null;
        elements.actionButton.textContent = action.label;
      }
    });
  }

  function installResourceDownloader() {
    if (!SETTINGS.downloadResources) return;
    const mount = () => {
      const action = getDownloadAction();
      if (!action || document.getElementById(DOWNLOAD_TOOL_ID)) return;
      installDownloadStyle();
      const elements = createDownloadTool(action);
      (document.body || document.documentElement).appendChild(elements.root);
      bindDownloadAction(action, elements);
    };
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', mount, { once: true });
    } else {
      mount();
    }
  }
